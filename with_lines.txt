9 const extractLockablePaths = (course: any) => {
10   const paths: any[] = [];
11   course.modules?.forEach((module: any) => {
12     const modObj = { moduleId: module.id, moduleTitle: module.title, arrays: [] as any[] };
13     Object.entries(module).forEach(([key, value]) => {
14       if (Array.isArray(value)) {
15         const arrObj = { key, items: [] as any[] };
16         value.forEach((item: any, idx: number) => {
17           let label = `Mục ${idx + 1}`;
18           if (typeof item === 'string') label = item.substring(0, 50) + (item.length > 50 ? '...' : '');
19           else if (item.name) label = item.name;
20           else if (item.title) label = item.title;
21           else if (item.group) label = item.group;
22           else if (item.groupName) label = item.groupName;
23           else if (item.task) label = item.task;
24           else if (item.problem) label = item.problem;
25           
26           arrObj.items.push({ path: `${module.id}.${key}.${idx}`, label });
27         });
28         modObj.arrays.push(arrObj);
29       }
30     });
31     if (modObj.arrays.length > 0) paths.push(modObj);
32   });
33   return paths;
34 };
